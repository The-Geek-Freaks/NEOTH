pub struct WalFollowArgs {
    #[arg(long, value_name = "DIR")]
    pub home: Option<PathBuf>,
    /// Comma-separated hex codes to emit, e.g. "0x40,0x7C,0xF4". Default: all.
    #[arg(long, value_name = "HEX_LIST")]
    pub types: Option<String>,
}

pub async fn run_wal_follow(mut args: WalFollowArgs) -> Result<()> {
    if args.home.is_none() {
        args.home = Some(crate::config::FreedomConfig::default_neoth_home());
    }
    let wal_dir = args.home.unwrap().join("wal");
    let filter: Option<std::collections::HashSet<u8>> = args.types.map(|s| {
        s.split(',').filter_map(|t| {
            let t = t.trim();
            if let Some(h) = t.strip_prefix("0x").or_else(|| t.strip_prefix("0X")) {
                u8::from_str_radix(h, 16).ok()
            } else {
                t.parse().ok()
            }
        }).collect()
    });

    // Track (path, byte_offset) per WAL segment.
    let mut cursors: std::collections::HashMap<PathBuf, usize> = Default::default();
    let stdout = std::io::stdout();
    loop {
        // Discover new *.wal files (handles segment rotation).
        let mut paths: Vec<_> = std::fs::read_dir(&wal_dir)?.flatten()
            .map(|e| e.path())
            .filter(|p| p.extension().and_then(|s| s.to_str()) == Some("wal"))
            .collect();
        paths.sort();
        for path in &paths {
            let offset = cursors.entry(path.clone()).or_insert(0);
            let Ok(bytes) = std::fs::read(path) else { continue };
            if bytes.len() <= *offset { continue }
            let slice = &bytes[*offset..];
            // Parse header once to get header_len for first read.
            let hdr_len = if *offset == 0 {
                crate::wal::segment_header::parse_segment_header(&bytes)
                    .map(|h| h.header_len()).unwrap_or(0)
            } else { 0 };
            let mut cur = if *offset == 0 { hdr_len } else { 0 };
            while cur < slice.len() {
                match crate::wal::frame::decode_frame(&slice[cur..]) {
                    Ok(dec) => {
                        let total = dec.header.total_len as usize;
                        let et = dec.header.event_type;
                        let sub = dec.header.event_subtype; // 0 for non-extended
                        if filter.as_ref().map_or(true, |f| f.contains(&et)) {
                            let row = serde_json::json!({
                                "event_type": et,
                                "subtype": sub,
                                "ts": serde_json::from_slice::<serde_json::Value>(dec.payload)
                                    .ok().and_then(|v| v.get("ts_unix").cloned())
                                    .unwrap_or(serde_json::Value::Null),
                            });
                            {
                                use std::io::Write;
                                let mut out = stdout.lock();
                                let _ = writeln!(out, "{row}");
                            }
                        }
                        if total == 0 { break }
                        cur = cur.saturating_add(total);
                    }
                    Err(_) => break, // torn tail — stop until more bytes arrive
                }
            }
            *offset += cur;
        }
        std::thread::sleep(std::time::Duration::from_millis(500));
    }
